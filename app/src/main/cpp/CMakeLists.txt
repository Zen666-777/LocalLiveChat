cmake_minimum_required(VERSION 3.22.1)
project(mel_llm LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS mel_compile=2)
set(CMAKE_JOB_POOL_COMPILE mel_compile)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_CURL LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_BACKEND_DL GGML_CPU_KLEIDIAI)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
# Optimized inference is needed even in an installable debug APK.
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
add_subdirectory(vendor/llama.cpp build-llama EXCLUDE_FROM_ALL)
add_library(mel_engine STATIC engine.cpp)
target_include_directories(mel_engine PUBLIC vendor/llama.cpp/include vendor/llama.cpp/ggml/include)
target_link_libraries(mel_engine PUBLIC llama)
if(ANDROID)
    add_library(mel_llm SHARED bridge.cpp)
    target_link_libraries(mel_llm mel_engine android log)
else()
    add_executable(mel-smoke smoke.cpp)
    target_link_libraries(mel-smoke mel_engine)
    find_package(JNI REQUIRED)
    add_library(mel_llm SHARED bridge.cpp)
    target_include_directories(mel_llm PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(mel_llm mel_engine)
endif()
